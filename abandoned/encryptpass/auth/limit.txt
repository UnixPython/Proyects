# FRAGMENTO QUE VA AQUÍ:

if os.path.exists(STATE_FILE):
    with open(STATE_FILE, "r") as f:
        fcntl.flock(f.fileno(), fcntl.LOCK_SH)   # lock compartido
        data = json.load(f)                        # lectura segura
else:
    data = {
        "fallos"        : [],
        "lockout_count" : 0,
        "bloqueado_hasta": 0.0,
    }

# POR QUÉ LOCK_SH aquí y no LOCK_EX:
# lectura → varios procesos pueden leer a la vez (LOCK_SH)
# escritura → solo uno a la vez (LOCK_EX) → eso va en _guardar_estado


# ──────────────────────────────────────────────────────────────
# _guardar_estado()
# ──────────────────────────────────────────────────────────────

# FRAGMENTO QUE VA AQUÍ:

tmp = STATE_FILE + ".tmp"

with open(tmp, "w") as f:
    fcntl.flock(f.fileno(), fcntl.LOCK_EX)
    json.dump(data, f)
    f.flush()
    os.fsync(f.fileno())

os.rename(tmp, STATE_FILE)
os.chmod(STATE_FILE, 0o600)

# POR QUÉ en este orden exacto:
# json.dump  → escribe en tmp, no en el original
# fsync      → garantiza que llegó al disco físico
# rename     → syscall atómica, swapea tmp → original
# chmod      → permisos restrictivos después del rename

# FRAGMENTO QUE VA EN ESE ARCHIVO:

class RateLimitErr(Exception):
    def __init__(
        self,
        tiempo_espera   : float,
        bloqueado_hasta : float,
        tier            : int,
    ):
        self.tiempo_espera   = tiempo_espera
        self.bloqueado_hasta = bloqueado_hasta
        self.tier            = tier
