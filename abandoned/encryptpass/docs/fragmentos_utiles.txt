import fcntl
import os
import time
from contextlib import contextmanager

@contextmanager
def file_lock(filepath):
    # Open the file; 'a+' or 'r+' avoids truncating the data while locking
    file_object = open(filepath, 'a+')
    try:
        # LOCK_EX: Exclusive lock
        # LOCK_NB: Non-blocking (raises exception instead of waiting)
        fcntl.lockf(file_object, fcntl.LOCK_EX | fcntl.LOCK_NB)
        yield file_object
    except (OSError, BlockingIOError):
        print(f"Process {os.getpid()} cannot acquire lock. File is busy.")
        raise
    finally:
        try:
            # LOCK_UN: Release the lock
            fcntl.lockf(file_object, fcntl.LOCK_UN)
        except OSError:
            pass
        file_object.close()

# Usage Example:
if __name__ == "__main__":
    try:
        with file_lock("my_app.lock") as lock_file:
            print("Lock acquired safely! Doing critical work...")
            time.sleep(5)
    except (OSError, BlockingIOError):
        print("Could not start. Another instance is running.")

from time import time, sleep


class SlidingWindow:

    def __init__(self, capacity, time_unit, forward_callback, drop_callback):
        self.capacity = capacity
        self.time_unit = time_unit
        self.forward_callback = forward_callback
        self.drop_callback = drop_callback

        self.cur_time = time()
        self.pre_count = capacity
        self.cur_count = 0

    def handle(self, packet):

        if (time() - self.cur_time) > self.time_unit:
            self.cur_time = time()
            self.pre_count = self.cur_count
            self.cur_count = 0

        ec = (self.pre_count * (self.time_unit - (time() - self.cur_time)) / self.time_unit) + self.cur_count

        if (ec > self.capacity):
            return self.drop_callback(packet)

        self.cur_count += 1
        return self.forward_callback(packet)
